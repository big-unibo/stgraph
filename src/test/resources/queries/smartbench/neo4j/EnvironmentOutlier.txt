MATCH (environment:Infrastructure)-[:hasCoverage]-(device:Sensor)-[:hasTemperature]-(temp:Temperature)
WHERE temp.timestamp >= datetime({epochMillis: {TSTAMPFROM_PARAM}}) AND temp.timestamp < datetime({epochMillis: {TSTAMPTO_PARAM}})
WITH environment, temp,
     // The WKT line of the environment (e.g., LINESTRING(34 0 0, 63 20 0)) as a list of [x, y]
     [c IN split(replace(replace(environment.location, 'LINESTRING(', ''), ')', ''), ',') | [v IN split(trim(c), ' ') WHERE v <> '' | toFloat(v)][0..2]] AS line,
     // The WKT point of the event (e.g., POINT(76 17 0), as loaded by SmartBenchNeo4J) as [x, y]
     [v IN split(trim(replace(replace(temp.location, 'POINT(', ''), ')', '')), ' ') WHERE v <> '' | toFloat(v)][0..2] AS p
// As ST_INTERSECTS of STGraph, PGAge and Graphix (2D): the event lies on a segment of the line of the environment
WHERE any(i IN range(0, size(line) - 2) WHERE
          (line[i + 1][0] - line[i][0]) * (p[1] - line[i][1]) - (line[i + 1][1] - line[i][1]) * (p[0] - line[i][0]) = 0
          AND p[0] >= CASE WHEN line[i][0] < line[i + 1][0] THEN line[i][0] ELSE line[i + 1][0] END
          AND p[0] <= CASE WHEN line[i][0] > line[i + 1][0] THEN line[i][0] ELSE line[i + 1][0] END
          AND p[1] >= CASE WHEN line[i][1] < line[i + 1][1] THEN line[i][1] ELSE line[i + 1][1] END
          AND p[1] <= CASE WHEN line[i][1] > line[i + 1][1] THEN line[i][1] ELSE line[i + 1][1] END)
WITH environment, AVG(temp.temperature) AS avgTemp
WHERE avgTemp > 50.5
RETURN DISTINCT environment;
