SELECT g.type, avg(m.value)
FROM (
    SELECT *
    FROM cypher('{GRAPH}'::name, $$
        MATCH (g:Segment)<-[:inSegment]-(v:StationVersion)<-[:hasVersion]-(:Station)-[:hasTimeSeries]->(t:TimeSeries)
        RETURN v.type, v.valid_from, v.valid_to, id(t)
    $$) AS (type agtype, valid_from bigint, valid_to bigint, tsid bigint)
) g
JOIN {GRAPH}.measurements m ON m.ts_id = g.tsid AND m.timestamp >= g.valid_from AND m.timestamp < g.valid_to
WHERE m.label = 'lane0_avg_speed'
GROUP BY g.type
