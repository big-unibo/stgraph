SELECT g.segment_id, avg(m.value)
FROM (
    SELECT *
    FROM cypher('{GRAPH}'::name, $$
        MATCH (g:Segment)<-[:inSegment]-(v:StationVersion)<-[:hasVersion]-(s:Station)-[:hasTimeSeries]->(t:TimeSeries)
        RETURN g.segment_id, v.valid_from, v.valid_to, id(t)
    $$) AS (segment_id agtype, valid_from bigint, valid_to bigint, tsid bigint)
) g
JOIN {GRAPH}.measurements m ON m.ts_id = g.tsid AND m.timestamp >= g.valid_from AND m.timestamp < g.valid_to AND m.timestamp >= {FROM} AND m.timestamp < {TO}
WHERE m.label = 'lane0_avg_speed'
GROUP BY g.segment_id
