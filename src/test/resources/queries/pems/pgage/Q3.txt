SELECT g.severity, avg(m.value)
FROM (
    SELECT *
    FROM cypher('{GRAPH}'::name, $$
        MATCH (i:Incident)-[:locatedIn]->(g:Segment)<-[:inSegment]-(v:StationVersion)<-[:hasVersion]-(:Station)-[:hasTimeSeries]->(t:TimeSeries)
        RETURN i.severity, i.valid_from, i.valid_to, v.valid_from, v.valid_to, id(t)
    $$) AS (severity agtype, i_from bigint, i_to bigint, valid_from bigint, valid_to bigint, tsid bigint)
) g
JOIN {GRAPH}.measurements m ON m.ts_id = g.tsid AND m.timestamp >= g.valid_from AND m.timestamp < g.valid_to
    AND m.timestamp >= g.i_from AND m.timestamp < g.i_to
WHERE m.label = 'lane0_avg_speed'
GROUP BY g.severity
