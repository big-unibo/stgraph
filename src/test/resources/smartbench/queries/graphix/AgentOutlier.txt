FROM (
  FROM GRAPH SmartBench
  MATCH (i:`Infrastructure`)<-[:`hasCoverage`]-(s:`Sensor`)-[:`hasTemperature`]->(ts:`Temperature`)
  SELECT s.id AS device, i.id AS environment, ts.id AS tsid
) AS g
JOIN `Observations` m ON m.tsid = g.tsid
WHERE m.`_id` >= {TSTAMPFROM_PARAM} AND m.`_id` < {TSTAMPTO_PARAM}
GROUP BY g.device AS device, g.environment AS environment
SELECT device, environment, MAX(m.`value`) AS maxValue;
