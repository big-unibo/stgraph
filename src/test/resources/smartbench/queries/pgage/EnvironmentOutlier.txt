WITH env_device AS (
    -- Each environment with the sensors that cover it (and have a time series), as STGraph
    SELECT *
    FROM cypher(
        'smartbench_graph', $$
            MATCH (t:Temperature)<-[:hasTemperature]-(d:Sensor)-[:hasCoverage]-(e:Infrastructure)
            RETURN DISTINCT e AS environment_node,
                            d.id AS device_id
        $$) AS (environment_node agtype, device_id text)
),
env_geom AS (
    SELECT
        environment_node,
        device_id,
        ST_GeomFromText(
            agtype_to_json(environment_node)->'properties'->>'location',
            4326
        ) AS geom
    FROM env_device
),
env_avg AS (
    SELECT
        e.environment_node,
        AVG(m.val) AS avg_temperature
    FROM env_geom e
    JOIN public.measurements m
        ON m.sensor_id = e.device_id
    WHERE m.time >= to_timestamp({TSTAMPFROM_PARAM} / 1000.0)
      AND m.time < to_timestamp({TSTAMPTO_PARAM} / 1000.0)
      AND ST_Intersects(m.location, e.geom)
    GROUP BY e.environment_node
)
SELECT environment_node
FROM env_avg
WHERE avg_temperature > 50.5;
